local function Create(class, props)
	local obj = Instance.new(class)
	for k, v in pairs(props) do
		if k ~= "Parent" then obj[k] = v end
	end
	if props.Parent then obj.Parent = props.Parent end
	return obj
end

local function Tween(obj, goal, t, style, dir)
	TweenService:Create(obj,
		TweenInfo.new(t or 0.15, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
		goal
	):Play()
end

local function Notify(msg)
	local sg = player.PlayerGui:FindFirstChild("AceHubGui")
	if not sg then return end
	local existing = sg:FindFirstChild("Notif")
	if existing then existing:Destroy() end

	local frame = Create("Frame", {
		Name             = "Notif",
		Size             = UDim2.new(0, 240, 0, 34),
		Position         = UDim2.new(0.5, -120, 1, -60),
		BackgroundColor3 = CONFIG.TitleBarColor,
		BorderSizePixel  = 0,
		BackgroundTransparency = 1,
		Parent           = sg,
	})
	Create("UICorner",  { CornerRadius = UDim.new(1, 0), Parent = frame })
	Create("UIStroke",  { Color = CONFIG.BorderColor, Thickness = 1, Parent = frame })
	Create("TextLabel", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text = msg,
		TextColor3 = CONFIG.TextColor,
		TextSize = 12,
		Font = Enum.Font.GothamBold,
		Parent = frame,
	})
	Tween(frame, { BackgroundTransparency = 0 }, 0.18)
	task.delay(2, function()
		Tween(frame, { BackgroundTransparency = 1 }, 0.2)
		task.wait(0.22)
		if frame and frame.Parent then frame:Destroy() end
	end)
end

local tabs      = {}

local function setActiveTab(name)
	for _, tname in ipairs(tabNames) do
		local btn   = tabs[tname]
		local panel = tabPanels[tname]
		local active = (tname == name)
		if btn then
			btn.TextColor3 = active and CONFIG.TextColor or CONFIG.MutedText
			local acc = btn:FindFirstChild("Accent")
			if acc then Tween(acc, { BackgroundTransparency = active and 0 or 1 }, 0.15) end
		end
		if panel then
			panel.Visible = active
		end
	end
end

-- Slider with manual TextBox input
local function AddSlider(panel, name, min, max, default, order, cb)
	local row = Create("Frame", {
		Size=UDim2.new(1,0,0,72), BackgroundColor3=CONFIG.SurfaceColor,
		BorderSizePixel=0, LayoutOrder=order or 0, Parent=panel,
	})
	Create("UICorner", { CornerRadius=UDim.new(0,7), Parent=row })
	Create("UIStroke", { Color=CONFIG.BorderColor, Thickness=1, Parent=row })

	Create("TextLabel", {
		Size=UDim2.new(0.55,0,0,20), Position=UDim2.new(0,12,0,8),
		BackgroundTransparency=1, Text=name, TextColor3=CONFIG.TextColor,
		TextSize=12, Font=Enum.Font.GothamBold, TextXAlignment=Enum.TextXAlignment.Left, Parent=row,
	})

	-- Manual input TextBox
	local inputBox = Create("TextBox", {
		Size             = UDim2.new(0,58,0,22),
		Position         = UDim2.new(1,-70,0,7),
		BackgroundColor3 = Color3.fromRGB(30,30,50),
		BorderSizePixel  = 0,
		Text             = tostring(default),
		TextColor3       = CONFIG.AccentColor,
		PlaceholderText  = "0",
		TextSize         = 12,
		Font             = Enum.Font.GothamBold,
		TextXAlignment   = Enum.TextXAlignment.Center,
		ClearTextOnFocus = false,
		Parent           = row,
	})
	Create("UICorner", { CornerRadius=UDim.new(0,5), Parent=inputBox })
	Create("UIStroke", { Color=CONFIG.AccentColor, Thickness=1, Transparency=0.5, Parent=inputBox })

	-- Track
	local trackBg = Create("Frame", {
		Size=UDim2.new(1,-24,0,4), Position=UDim2.new(0,12,0,44),
		BackgroundColor3=Color3.fromRGB(45,45,65), BorderSizePixel=0, Parent=row,
	})
	Create("UICorner", { CornerRadius=UDim.new(1,0), Parent=trackBg })

	local fill = Create("Frame", {
		Size=UDim2.new((default-min)/(max-min),0,1,0),
		BackgroundColor3=CONFIG.AccentColor, BorderSizePixel=0, Parent=trackBg,
	})
	Create("UICorner", { CornerRadius=UDim.new(1,0), Parent=fill })

	local knob = Create("Frame", {
		Size=UDim2.new(0,14,0,14), AnchorPoint=Vector2.new(0.5,0.5),
		Position=UDim2.new((default-min)/(max-min),0,0.5,0),
		BackgroundColor3=CONFIG.AccentColor, BorderSizePixel=0, Parent=trackBg,
	})
	Create("UICorner", { CornerRadius=UDim.new(1,0), Parent=knob })

	local value   = default
	local sliding = false

	local function applyValue(v)
		v = math.clamp(math.floor(v), min, max)
		value = v
		local rel = (v - min) / (max - min)
		fill.Size     = UDim2.new(rel, 0, 1, 0)
		knob.Position = UDim2.new(rel, 0, 0.5, 0)
		inputBox.Text = tostring(v)
		if cb then cb(v) end
	end

	local function updateFromX(inputX)
		local pos = trackBg.AbsolutePosition
		local sz  = trackBg.AbsoluteSize
		local rel = math.clamp((inputX - pos.X) / sz.X, 0, 1)
		applyValue(min + (max - min) * rel)
	end

	-- TextBox submit
	inputBox.FocusLost:Connect(function()
		local num = tonumber(inputBox.Text)
		if num then
			applyValue(num)
		else
			inputBox.Text = tostring(value)
		end
	end)

	-- Slider drag
	local hitArea = Create("TextButton", {
		Size=UDim2.new(1,0,1,0), BackgroundTransparency=1, Text="", ZIndex=5, Parent=trackBg,
	})
	hitArea.MouseButton1Down:Connect(function() sliding = true end)
	hitArea.InputBegan:Connect(function(inp)
		if inp.UserInputType == Enum.UserInputType.Touch then sliding = true end
	end)
	UserInputService.InputChanged:Connect(function(inp)
		if not sliding then return end
		if inp.UserInputType == Enum.UserInputType.MouseMovement
			or inp.UserInputType == Enum.UserInputType.Touch then
			updateFromX(inp.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(inp)
		if inp.UserInputType == Enum.UserInputType.MouseButton1
			or inp.UserInputType == Enum.UserInputType.Touch then
			sliding = false
		end
	end)

	return row, function() return value end, applyValue
end

	local applyBtn = Create("TextButton", {
		Size=UDim2.new(0,70,0,26), Position=UDim2.new(1,-84,0.5,-13),

			Notify("Invalid number for "..toolName)
			return
		end

			or (player.Character and player.Character:FindFirstChild(toolName))
		if tool then
			tool:SetAttribute("Count", num)
			Notify(toolName.." → "..num)
		else
			Notify("Tool not found: "..toolName)
		end
	end)

	return card
end

-- ════════════════════════════════════════════
--   SCAN BACKPACK & BUILD MAIN TAB
-- ════════════════════════════════════════════
local function ScanAndBuildMain()
	local p = tabPanels["Main"]
	-- Clear old children (except UIListLayout)
	for _, c in ipairs(p:GetChildren()) do
		if c:IsA("GuiObject") then c:Destroy() end
	end

	local gearItems, seedItems = {}, {}

	local function checkTool(tool)
		if not tool:IsA("Tool") then return end
		local cat   = tool:GetAttribute("MainCategory")
		local count = tool:GetAttribute("Count") or 0
		if cat == "Gear" then
			table.insert(gearItems, { name=tool.Name, count=count })
		elseif cat == "Seed" then
			table.insert(seedItems, { name=tool.Name, count=count })
		end
	end

	for _, tool in ipairs(backpack:GetChildren()) do checkTool(tool) end
	if player.Character then
		for _, tool in ipairs(player.Character:GetChildren()) do checkTool(tool) end
	end

	local order = 1

	if #gearItems > 0 then
		AddLabel(p, "Gear", order) ; order += 1
		for _, item in ipairs(gearItems) do
			AddToolCard(p, item.name, "Gear", item.count, order) ; order += 1
		end
	end

	if #seedItems > 0 then
		if #gearItems > 0 then AddDivider(p, order) ; order += 1 end
		AddLabel(p, "Seeds", order) ; order += 1
		for _, item in ipairs(seedItems) do
			AddToolCard(p, item.name, "Seed", item.count, order) ; order += 1
		end
	end

	if #gearItems == 0 and #seedItems == 0 then
		Create("TextLabel", {
			Size=UDim2.new(1,0,0,60), BackgroundTransparency=1,
			Text="No tools found in Backpack.\nEquip some gear or seeds first.",
			TextColor3=CONFIG.MutedText, TextSize=12,
			Font=Enum.Font.Gotham, TextWrapped=true,
			LayoutOrder=1, Parent=p,
		})
	end

	AddDivider(p, order) ; order += 1
	AddButton(p, "↺  Refresh Backpack", Color3.fromRGB(35,35,55), order, function()
		ScanAndBuildMain()
		Notify("Backpack refreshed!")
	end)
end

-- ════════════════════════════════════════════
--   DRAGGING (PC + Mobile)
-- ════════════════════════════════════════════
local dragging, dragStart, startPos = false, nil, nil

titleBar.InputBegan:Connect(function(inp)
	if inp.UserInputType == Enum.UserInputType.MouseButton1
		or inp.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = inp.Position
		startPos  = mainFrame.Position
	end
end)
UserInputService.InputChanged:Connect(function(inp)
	if not dragging then return end
	if inp.UserInputType == Enum.UserInputType.MouseMovement
		or inp.UserInputType == Enum.UserInputType.Touch then
		local d = inp.Position - dragStart
		mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+d.X,
			startPos.Y.Scale, startPos.Y.Offset+d.Y)
	end
end)
UserInputService.InputEnded:Connect(function(inp)
	if inp.UserInputType == Enum.UserInputType.MouseButton1
		or inp.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)
